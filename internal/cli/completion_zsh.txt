#compdef luuna-missions lms lm

_luuna_missions() {
    local -a commands
    commands=(
        'init:Initialize .luuna/missions/ in current directory'
        'create:Create a new issue'
        'show:Show one or more issues'
        'list:List issues with filters'
        'ready:Show open issues with no unresolved blockers'
        'search:Full-text search title + description'
        'update:Update issue fields'
        'close:Close one or more issues'
        'dep:Manage issue dependencies'
        'label:Manage issue labels'
        'blocked:Show all blocked issues'
        'stats:Project statistics'
        'sync:Stage and commit .luuna/missions/ changes'
        'doctor:Check project health and data integrity'
        'tpl:Manage issue templates (molecules)'
        'migrate-from-beads:Migrate issues from beads issue tracker'
        'prime:Output AI agent context'
        'onboard:Add missions section to CLAUDE.md / AGENTS.md'
        'upgrade:Update Luuna Missions through a reviewed checkout'
        'completions:Output shell completion script'
        'block:Add a blocker to an issue'
        'unblock:Remove blockers from an issue'
        'plan:Plan management'
        'config:Read, write, and inspect .luuna/missions/config.yaml'
        'setup:Install Claude Code hooks: prime on session start, guard .luuna/missions/*.jsonl edits'
        'guard:PreToolUse hook: deny hand edits to .luuna/missions/*.jsonl (reads hook JSON on stdin)'
    )

    _arguments -s \
        '1:command:_describe "command" commands' \
        '*::arg:->args'

    case "$state" in
    args)
    case "$words[1]" in
    init)
        _arguments -s \
        '--json[Output as JSON]'
        ;;
    create)
        _arguments -s \
        '--title[Issue title]' \
        '--type[Issue type (task|bug|feature|epic)]' \
        '--priority[Priority 0-4 or P0-P4]' \
        '--assignee[Assignee name]' \
        '--description[Issue description]' \
        '--desc[Issue description (alias for --description)]' \
        '--body[Issue description (alias for --description)]' \
        '--labels[Comma-separated labels]' \
        '--label[Comma-separated labels (alias)]' \
        '--from[Record the issue this was discovered from (provenance, non-blocking)]' \
        '--intent[Link to a roots idea (repeatable)]' \
        '--json[Output as JSON]'
        ;;
    show)
        _arguments -s \
        '--format[Output format (markdown|compact|plain|ids|json)]' \
        '--json[Output as JSON (alias for --format json)]'
        ;;
    list)
        _arguments -s \
        '--status[Filter by status (open|in_progress|closed)]' \
        '--type[Filter by type (task|bug|feature|epic)]' \
        '--assignee[Filter by assignee]' \
        '--all[Include closed issues (default: only open/in_progress)]' \
        '--label[Filter: must have ALL labels (comma-separated, AND)]' \
        '--label-any[Filter: must have any label (comma-separated, OR)]' \
        '--unlabeled[Filter: issues with no labels]' \
        '--priority[Filter by priority (comma-separated, e.g. 0,1 or P0,P1)]' \
        '--priority-max[Filter to priority <= n (e.g. --priority-max 1 = P0+P1)]' \
        '--limit[Max issues to show]' \
        '--sort[Sort order (priority|created|updated|id)]' \
        '--format[Output format (markdown|compact|plain|ids|json)]' \
        '--json[Output as JSON (alias for --format json)]'
        ;;
    ready)
        _arguments -s \
        '--type[Filter by type (task|bug|feature|epic)]' \
        '--assignee[Filter by assignee]' \
        '--label[Filter: must have ALL labels (comma-separated, AND)]' \
        '--label-any[Filter: must have any label (comma-separated, OR)]' \
        '--unlabeled[Filter: issues with no labels]' \
        '--priority[Filter by priority (comma-separated, e.g. 0,1 or P0,P1)]' \
        '--priority-max[Filter to priority <= n (e.g. --priority-max 1 = P0+P1)]' \
        '--limit[Max issues to show]' \
        '--sort[Sort order (priority|created|updated|id)]' \
        '--format[Output format (markdown|compact|plain|ids|json)]' \
        '--json[Output as JSON (alias for --format json)]' \
        '--respect-schedule[Exclude issues with extensions.queued=true or future extensions.scheduledFor]'
        ;;
    search)
        _arguments -s \
        '--status[Filter by status (open|in_progress|closed)]' \
        '--type[Filter by type (task|bug|feature|epic)]' \
        '--assignee[Filter by assignee]' \
        '--label[Filter: must have ALL labels (comma-separated, AND)]' \
        '--label-any[Filter: must have any label (comma-separated, OR)]' \
        '--unlabeled[Filter: issues with no labels]' \
        '--priority[Filter by priority (comma-separated, e.g. 0,1 or P0,P1)]' \
        '--priority-max[Filter to priority <= n (e.g. --priority-max 1 = P0+P1)]' \
        '--limit[Max issues to show]' \
        '--sort[Sort order (priority|created|updated|id)]' \
        '--format[Output format (markdown|compact|plain|ids|json)]' \
        '--json[Output as JSON (alias for --format json)]'
        ;;
    update)
        _arguments -s \
        '--status[New status (open|in_progress|closed)]' \
        '--title[New title]' \
        '--assignee[New assignee]' \
        '--description[New description]' \
        '--desc[New description (alias for --description)]' \
        '--body[New description (alias for --description)]' \
        '--type[New type (task|bug|feature|epic)]' \
        '--priority[New priority 0-4 or P0-P4]' \
        '--add-label[Add label(s) (comma-separated)]' \
        '--remove-label[Remove label(s) (comma-separated)]' \
        '--set-labels[Set labels (comma-separated, empty to clear)]' \
        '--extensions[Shallow-merge JSON object into Issue.extensions]' \
        '--clear-extensions[Remove the extensions field]' \
        '--claim[Atomically claim: open → in_progress, assignee = --as (fails if taken)]' \
        '--as[Identity for --claim (default: $USER)]' \
        '--intent[Set linked roots idea(s) (repeatable, replaces)]' \
        '--json[Output as JSON]'
        ;;
    close)
        _arguments -s \
        '--reason[Close reason]' \
        '--json[Output as JSON]'
        ;;
    dep)
        _arguments -s \
        '1:subcommand:(('
            'add:Add a dependency (issue depends on depends-on)'
            'remove:Remove a dependency'
            'list:Show dependencies for an issue'
        ))'
        ;;
    label)
        _arguments -s \
        '1:subcommand:(('
            'add:Add labels to an issue'
            'remove:Remove labels from an issue'
            'list:List labels on an issue'
            'list-all:List all labels used in the project'
        ))'
        ;;
    blocked)
        _arguments -s \
        '--format[Output format (markdown|compact|plain|ids|json)]' \
        '--json[Output as JSON (alias for --format json)]'
        ;;
    stats)
        _arguments -s \
        '--format[Output format (markdown|compact|plain|ids|json)]' \
        '--json[Output as JSON (alias for --format json)]'
        ;;
    sync)
        _arguments -s \
        '--status[Check status without committing]' \
        '--dry-run[Show what would be committed without committing]' \
        '--json[Output as JSON]'
        ;;
    doctor)
        _arguments -s \
        '--fix[Auto-fix fixable issues]' \
        '--verbose[Show all check results including passes]' \
        '--json[Output as JSON]'
        ;;
    tpl)
        _arguments -s \
        '1:subcommand:(('
            'create:Create a new template'
            'step:Manage template steps'
            'list:List all templates'
            'show:Show template with steps'
            'pour:Instantiate template into issues'
            'status:Show convoy status for a template'
        ))'
        ;;
    migrate-from-beads)
        _arguments -s \
        '--json[Output as JSON]'
        ;;
    prime)
        _arguments -s \
        '--compact[Condensed quick-reference output]' \
        '--export[Output the default template]' \
        '--json[Output as JSON]'
        ;;
    onboard)
        _arguments -s \
        '--stdout[Print what would be written to stdout]' \
        '--check[Check status without modifying files]' \
        '--json[Output as JSON]'
        ;;
    upgrade)
        _arguments -s \
        '--check[Check for updates without installing]' \
        '--json[Output as JSON]'
        ;;
    block)
        _arguments -s \
        '--by[Issue that blocks this issue]' \
        '--json[Output as JSON]'
        ;;
    unblock)
        _arguments -s \
        '--from[Remove a specific blocker]' \
        '--all[Remove all closed blockers]' \
        '--json[Output as JSON]'
        ;;
    plan)
        _arguments -s \
        '1:subcommand:(('
            'templates:List available plan templates'
            'prompt:Emit structured planning prompt JSON for a seed'
            'submit:Validate a plan, spawn children, write plans.jsonl row'
            'show:Show a plan with sections, children, and status (accepts plan id or seed id)'
            'validate:Re-run validation against the current template definition (accepts plan id or seed id)'
            'outcome:Record a plan outcome (storage-only; not a state transition; accepts plan id or seed id)'
            'review:Record a reviewer (informational; not a state transition; accepts plan id or seed id)'
            'edit:Edit plan fields in place (accepts plan id or seed id); bumps revision'
            'create:Create an adopt-only plan with zero spawned children (populate via '\''luuna-missions plan adopt'\'')'
            'adopt:Adopt existing open missions into a plan (link-only; bumps plan revision)'
            'reorder:Set the exact order of plan.children (must be a permutation of current children)'
            'release:Release missions from a plan (link-only; missions stay open; bumps plan revision)'
            'list:List plans with optional filters'
        ))'
        ;;
    config)
        _arguments -s \
        '1:subcommand:(('
            'schema:Emit the JSON Schema for .luuna/missions/config.yaml'
            'show:Print the current config (or a value at --path)'
            'set:Set a config value at <path>; <value> is YAML-parsed'
            'unset:Remove the config value at <path>'
        ))'
        ;;
    setup)
        _arguments -s \
        '--remove[Remove missions hooks]' \
        '--json[Output as JSON]'
        ;;
    esac
        ;;
    esac
}

_luuna_missions "$@"
